# Função de playlist otimizada para caminhos Windows e aleatoriedade via memória [cite: 1851, 1916]
def imc_playlist(diretorio) =
  arquivos = file.ls(diretorio)
  arquivos_imc = list.filter(fun (f) -> string.match(pattern="\\.imc$", f), arquivos)
  
  arquivos_com_url = list.map(fun (f) -> begin
    path_raw = "C:/liquidsoap-2.4.0-win64/#{diretorio}/#{f}"
    # Substitui barras invertidas para evitar o erro 'illegal backslash' [cite: 1891, 1915]
    path_clean = string.replace(pattern='\\\\', (fun (_) -> "/"), path_raw)
    path_encoded = url.encode(path_clean)
    
    # URL para o seu servidor C# descriptografador na porta 5005 [cite: 1915]
    endereco_url = "http://localhost:5005/?path=#{path_encoded}&ext=.mp3"
    "annotate:caminho_original=\"#{path_clean}\",normalize_url=false:#{endereco_url}"
  end, arquivos_imc)
  
  # Embaralhamento garantido antes de carregar na engine [cite: 1851]
  playlist.list(id=diretorio, list.shuffle(arquivos_com_url))
end

# --- 3. MIXAGEM E TRANSIÇÕES ---
def transition(a, b) =
  add([fade.initial(duration=0.3, b), fade.final(duration=0.3, a)])
end


# Rotate garante a proporção 3x3 entre músicas e comerciais [cite: 2008, 2041]
base_programacao = rotate(id="main_logic", weights=[3, 3], [ musics_grid, advertisements ])


# Remove silêncios excessivos entre as faixas [cite: 2035]
base_clean = blank.skip(max_blank=2.0, base_programacao)


# Fallback principal: Live > Eventos > Programação Musical [cite: 1296]
imc_audio = fallback(track_sensitive=false, transitions=[transition, transition, transition], [live, events, base_clean])


# --- 4. MAPEAMENTO DE METADADOS ---
def tratar_meta_url(m) =
  referencia = if m["caminho_original"] != "" then m["caminho_original"] else m["filename"] end
  
  # Extração do diretório pai para identificar o gênero no log [cite: 949]
  pattern = "[/\\\\]([^/\\\\]+)[/\\\\][^/\\\\]+$"
  extracao = string.extract(pattern=pattern, referencia)
  diretorio_pai = if list.length(extracao) > 0 then extracao[1] else "GERAL" end


  if m["caminho_original"] != "" then
    [("type", "music"), ("sub_type", diretorio_pai), ("filename", referencia)]
  else
    t = if m["type"] != "" then m["type"] else "commercial" end
    [("type", t), ("sub_type", diretorio_pai), ("filename", referencia)]
  end
end


imc_audio = metadata.map(tratar_meta_url, imc_audio)


# --- 5. SISTEMA DE LOG PERSONALIZADO (PLAYSTART) ---
# Formato: [PLAYSTART] [FONTE] [DIRETORIO] Nome do Arquivo [cite: 1907, 1927]
def log_inicio(m) =
  arquivo = path.basename(m["filename"])
  if arquivo != "" and arquivo != "." then
    fonte = string.uppercase(m["type"])
    pasta = string.uppercase(m["sub_type"])
    log(level=3, "[PLAYSTART] [#{fonte}] [#{pasta}] #{arquivo}")
  end
end


# Vinculação assíncrona obrigatória na v2.4.0 [cite: 1941, 1963]
imc_audio.on_track(log_inicio, synchronous=false)


# --- 6. API HTTP (COMUNICAÇÃO COM O PLAYER C#) ---
var_volume = ref(1.0)
audio_com_volume = amplify(var_volume, imc_audio)


# Processamento final para volume constante na loja [cite: 1281, 1303]
audio_processado = normalize(target=-1.0, threshold=-40.0, audio_com_volume)
audio_final = compress(threshold=-12.0, ratio=3.0, audio_processado)


meta = ref([])
imc_audio.on_metadata(synchronous=true, fun (m) -> meta.set(m))


# Endpoint para metadados e tempo restante [cite: 1299, 1808]
harbor.http.register(port=7000, method="GET", "/getmeta", 
  fun (_, res) -> begin
    d = source.duration(imc_audio)
    e = source.elapsed(imc_audio)
    remaining = if float.is_nan(d-e) then 0.0 else d-e end
    full_meta = list.add(("remaining_seconds", "#{remaining}"), meta())
    res.json(full_meta)
  end
)


# Endpoint para controle de volume remoto [cite: 1300, 1809]
harbor.http.register(port=7000, method="GET", "/set_volume", 
  fun (req, res) -> begin
    try
      v = float_of_string(list.assoc("level", req.query))
      var_volume.set(v)
      res.json([("status", "success"), ("level", "#{v}")])
    catch _ do
      res.json([("status", "error")])
    end
  end
)


# --- 7. SAÍDA FINAL ---
output.ao(fallible=true, audio_final)
